    function switchTab(tab) {
      const dash = document.getElementById('tabDashboard');
      const track = document.getElementById('tabTracker');
      const btn1 = document.getElementById('tabBtn1');
      const btn2 = document.getElementById('tabBtn2');

      const activeClass = ['bg-white', 'text-teal-700', 'shadow-sm', 'font-bold'];
      const inactiveClass = ['text-slate-500', 'hover:text-slate-800', 'hover:bg-white/50', 'font-medium'];

      if (tab === 'dashboard') {
        dash.classList.remove('hidden'); track.classList.add('hidden');
        btn1.classList.add(...activeClass); btn1.classList.remove(...inactiveClass);
        btn2.classList.remove(...activeClass); btn2.classList.add(...inactiveClass);
      } else {
        dash.classList.add('hidden'); track.classList.remove('hidden');
        btn2.classList.add(...activeClass); btn2.classList.remove(...inactiveClass);
        btn1.classList.remove(...activeClass); btn1.classList.add(...inactiveClass);
      }
    }
    
    function loadData(forceRefresh = false) {
      const year = document.getElementById('yearFilter').value;
      const month = document.getElementById('monthFilter').value;
      
      const refreshBtn = document.getElementById('refreshBtn');
      if (refreshBtn) {
        refreshBtn.innerHTML = '<svg class="animate-spin -ml-1 mr-2 h-4 w-4 text-white" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg> โหลดข้อมูล...';
        refreshBtn.disabled = true;
      }

      google.script.run
        .withSuccessHandler(function(res) {
          renderDashboard(res);
          if (refreshBtn) {
            refreshBtn.innerHTML = '<svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15"></path></svg> รีเฟรชข้อมูล';
            refreshBtn.disabled = false;
          }
        })
        .getDashboardData(year, month, forceRefresh);
    }

    function renderDashboard(res) {
      if (!res) return;
      document.getElementById('kpiVisits').innerText = res.kpis.totalVisits;
      document.getElementById('kpiPatients').innerText = res.kpis.uniquePatients;
      document.getElementById('kpiAvgINR').innerText = res.kpis.avgINR;

    function handleSearch() {
      const hnInput = document.getElementById('hnInput');
      const hn = hnInput.value.trim();
      const msg = document.getElementById('searchMsg');
      const card = document.getElementById('patientCard');
      const searchBtn = document.getElementById('searchBtn');

      if (!hn) {
        msg.innerText = '⚠️ กรุณากรอกรหัส HN ก่อนทำการค้นหา';
        msg.classList.remove('hidden', 'text-teal-600');
        msg.classList.add('text-rose-500');
        return;
      }
      
      msg.classList.add('hidden');
      hnInput.disabled = true;
      searchBtn.disabled = true;
      searchBtn.classList.replace('bg-teal-600', 'bg-teal-400');
      searchBtn.innerHTML = `<svg class="animate-spin h-5 w-5 text-white" fill="none" viewBox="0 0 24 24"><circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle><path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path></svg>`;

      const resetUI = () => {
        hnInput.disabled = false;
        searchBtn.disabled = false;
        searchBtn.classList.replace('bg-teal-400', 'bg-teal-600');
        searchBtn.innerHTML = 'ค้นหา';
      };

      google.script.run
        .withSuccessHandler(function(res) {
          resetUI();
          if (!res.success) {
            msg.innerText = '❌ ' + res.message;
            msg.classList.remove('hidden', 'text-teal-600');
            msg.classList.add('text-rose-500');
            card.classList.add('hidden');
            return;
          }

          document.getElementById('ptName').innerText = res.name || 'ไม่ระบุชื่อ';
          document.getElementById('ptHN').innerText = res.hn;
          document.getElementById('ptVisitCount').innerText = res.records.length;

          const tbody = document.getElementById('ptTableBody');
          tbody.innerHTML = '';
          const chartLabels = [];
          const chartINR = [];

          res.records.forEach((r, idx) => {
            chartLabels.push(r.date || ('ครั้งที่ ' + (idx + 1)));
            chartINR.push(r.inr);

            let badge = '';
            if (r.inr !== null) {
              if (r.inr < 2.0) badge = '<span class="bg-amber-100 text-amber-700 border border-amber-200 px-3 py-1 rounded-full text-xs font-bold">ต่ำกว่าเกณฑ์</span>';
              else if (r.inr <= 3.5) badge = '<span class="bg-emerald-100 text-emerald-700 border border-emerald-200 px-3 py-1 rounded-full text-xs font-bold">อยู่ในเกณฑ์ (2.0-3.5)</span>';
              else badge = '<span class="bg-rose-100 text-rose-700 border border-rose-200 px-3 py-1 rounded-full text-xs font-bold">สูงกว่าเกณฑ์</span>';
            } else {
              badge = '<span class="bg-slate-100 text-slate-500 border border-slate-200 px-3 py-1 rounded-full text-xs font-bold">ไม่มีผล</span>';
            }

            tbody.innerHTML += `
              <tr class="hover:bg-slate-50 transition-colors">
                <td class="px-8 py-4 font-semibold text-slate-800">${r.date}</td>
                <td class="px-8 py-4 text-slate-500 text-xs">${r.month || '-'} (ปี ${r.fy || '-'})</td>
                <td class="px-8 py-4 text-center font-mono font-extrabold text-slate-800 text-lg">${r.inr !== null ? r.inr.toFixed(2) : '-'}</td>
                <td class="px-8 py-4 text-right">${badge}</td>
              </tr>
            `;
          });

          card.classList.remove('hidden');

          if (patientChartInstance) patientChartInstance.destroy();
          const ctxP = document.getElementById('patientChart').getContext('2d');
          patientChartInstance = new Chart(ctxP, {
            type: 'line',
            data: {
              labels: chartLabels,
              datasets: [
                { label: 'ระดับ INR', data: chartINR, borderColor: '#0D9488', backgroundColor: 'rgba(13, 148, 136, 0.1)', fill: true, borderWidth: 3, tension: 0.3, pointBackgroundColor: '#fff', pointBorderWidth: 2, pointRadius: 5 },
                { label: 'เกณฑ์บน (3.5)', data: Array(chartLabels.length).fill(3.5), borderColor: '#F43F5E', borderDash: [5, 5], borderWidth: 2, pointRadius: 0, fill: false },
                { label: 'เกณฑ์ล่าง (2.0)', data: Array(chartLabels.length).fill(2.0), borderColor: '#F59E0B', borderDash: [5, 5], borderWidth: 2, pointRadius: 0, fill: false }
              ]
            },
            options: { 
              responsive: true, maintainAspectRatio: false, 
              scales: { x: { grid: { display: false } }, y: { min: 0, suggestedMax: 5, border: { dash: [4, 4] }, grid: { color: '#f1f5f9' } } }, 
              plugins: { 
                legend: { position: 'top', align: 'end' },
                datalabels: {
                  anchor: 'end',
                  align: 'top',
                  color: '#0D9488',
                  font: { family: 'Kanit', weight: 'bold', size: 10 },
                  formatter: (value, context) => context.datasetIndex === 0 && value !== null ? value.toFixed(2) : '' // แสดงเลขเฉพาะเส้นหลัก (ระดับ INR)
                }
              } 
            }
          });
        })
        .withFailureHandler(function(error) {
          resetUI();
          msg.innerText = '❌ เกิดข้อผิดพลาดในการเชื่อมต่อ: ' + error.message;
          msg.classList.remove('hidden', 'text-teal-600');
          msg.classList.add('text-rose-500');
        })
        .searchPatientByHN(hn);
    }
